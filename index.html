package controller;

import common.Korisnik;
import communication.Operation;
import communication.Receiver;
import communication.Request;
import communication.Response;
import communication.Sender;
import java.io.IOException;
import java.net.Socket;

public class Controller {
    private Socket socket;
    private static Controller instance;
    private Sender sender;
    private Receiver receiver;
    
    private Controller() throws IOException {
            socket = new Socket("localhost",9000);
            sender = new Sender(socket);
            receiver = new Receiver(socket);

    }
    
    public static Controller getInstance() throws IOException{
        if(instance==null)
        {
            instance = new Controller();
        }
        return instance;
    }
    
    public Object obradi(Operation operation, String argument)
    {
        try {
            Request request = new Request(operation, argument);
            sender.send(request);
            Response response = (Response)receiver.receive();
            
            if(response.getException() == null)
                return (String)response.getResult();
            else
                throw response.getException();
        } catch (Exception ex) {
            System.getLogger(Controller.class.getName()).log(System.Logger.Level.ERROR, (String) null, ex);
        }
        return null;
    }
}

package threads;

import common.Korisnik;
import communication.*;
import java.net.Socket;
import logic.Controller;

public class ClientHandler extends Thread{
    private Socket socket;
    private Receiver receiver;
    private Sender sender;
    private Korisnik user;
    
    public ClientHandler(Socket socket) {
        this.socket = socket;
        receiver = new Receiver(socket);
        sender = new Sender(socket);
        
    }

    @Override
    public void run() {
           
        while(!socket.isClosed())
        {
            try {
                Request request = (Request) receiver.receive();
                Response response = handleRequest(request);
                sender.send(response);
            } catch (Exception ex) {
                System.getLogger(ClientHandler.class.getName()).log(System.Logger.Level.ERROR, (String) null, ex);
            }
        }
    }

    private Response handleRequest(Request request) {
        Response response = new Response();
        String odgovor;
        String argument;
        try{
            switch(request.getOperation())
            {
                case LOGIN:
                    String username = (String) request.getArgument();
                    odgovor = Controller.getInstance().login(username);
                    this.user = new Korisnik(username);
                    response.setResult(odgovor);
                    break;
                case ADD:
                    argument = (String) request.getArgument();
                    odgovor = Controller.getInstance().saberi(argument);
                    response.setResult(odgovor);
                    break;
                case MINMAX:
                    argument = (String) request.getArgument();
                    odgovor = Controller.getInstance().minmax(argument);
                    response.setResult(odgovor);
                    break;
                case DATE:
                    odgovor = Controller.getInstance().date();
                    response.setResult(odgovor);
                    break;
                case LOGOUT:
                    if (this.user == null) {
                        throw new Exception("Korisnik nije ulogovan");
                    }
                    odgovor = Controller.getInstance().logout(this.user.getKorisnickoIme());
                    this.user = null;
                    response.setResult(odgovor);
                    break;
                default:
                    throw new AssertionError();
            }
        }
        catch(Exception ex){
            response.setException(ex);
        }
        
        return response;
    }
    
    public Socket getSocket(){
        return socket;
    }

    public Korisnik getUser() {
        return user;
    }
    
    
}

package threads;

import common.Korisnik;
import java.io.IOException;
import java.net.ServerSocket;
import java.net.Socket;
import java.util.ArrayList;
import java.util.List;

public class ServerThread extends Thread{
    private ServerSocket serverSocket;
    private List clients;
    
    public ServerThread() throws IOException {
        serverSocket = new ServerSocket(9000);
        clients = new ArrayList<>();
    }
    
    @Override
    public void run() {
            while(!serverSocket.isClosed())
            {
                try {
                    System.out.println("Cekam clienta");
                    Socket clientSocket = serverSocket.accept();
                    
                    ClientHandler thread = new ClientHandler(clientSocket);
                    thread.start();
                    clients.add(thread);
                    System.out.println("Klijent povezan");
                    
                } catch (IOException ex) {
                    System.getLogger(ServerThread.class.getName()).log(System.Logger.Level.ERROR, (String) null, ex);
                }
            }
            
            StopAllThreads();

    }
    
    private void StopAllThreads()
    {
        for(ClientHandler client : clients)
        {
            try {
                client.getSocket().close();
            } catch (IOException ex) {
                System.getLogger(ServerThread.class.getName()).log(System.Logger.Level.ERROR, (String) null, ex);
            }
        }
    }

    public ServerSocket getServerSocket() {
        return serverSocket;
    }
    
    private List getAllUsers()
    {
        List users = new ArrayList<>();
        
        for(ClientHandler client : clients)
        {
            users.add(client.getUser());
        }
        
        return users;
    }
}
